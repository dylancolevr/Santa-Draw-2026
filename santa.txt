  // Loads the reply through a script tag (JSONP). Works on phones and in-app browsers where fetch is blocked.
  function apiJsonp(params) {
    return new Promise(function (resolve, reject) {
      var cb = "cb" + Date.now() + Math.floor(Math.random() * 1e6), done = false, sc = document.createElement("script");
      function fin() { done = true; clearTimeout(t); try { delete window[cb]; } catch (e) { window[cb] = undefined; } if (sc.parentNode) sc.parentNode.removeChild(sc); }
      var t = setTimeout(function () { if (!done) { fin(); reject(new Error("timeout")); } }, 30000);
      window[cb] = function (d) { if (done) return; fin(); resolve(d); };
      sc.onerror = function () { if (done) return; fin(); reject(new Error("network")); };
      params.callback = cb;
      var q = Object.keys(params).map(function (k) { return k + "=" + encodeURIComponent(params[k]); }).join("&");
      sc.src = SCRIPT_URL + "?" + q;
      document.head.appendChild(sc);
    });
  }

  function apiFetch(params) {

  }

  function api(params) {

        var err = new Error((e1 && e1.message || "x") + "/" + (e2 && e2.message || "x"));
        throw err;
      });
    });
  }

  }

      renderGrid();
    }).catch(function () {});
  }
